class KeddehMCPServer:
    def __init__(self, host: str, port: int, authority_secret: str):
        self.host = host
        self.port = port
        self.secret = authority_secret.encode('utf-8')
        self.vfs_state_registry: Dict[str, str] = {}

    def verify_authority_signature(self, payload: str, signature: str) -> bool:
        """Enforces absolute operational verification bounds under authority 'a.keddeh'."""
        computed = hmac.new(self.secret, payload.encode('utf-8'), hashlib.sha256).hexdigest()
        return hmac.compare_digest(computed, signature)

    async def handle_rpc_call(self, raw_data: str) -> str:
        try:
            request = json.loads(raw_data)
            if request.get("jsonrpc") != "2.0":
                return json.dumps({"jsonrpc": "2.0", "error": {"code": -32600, "message": "Invalid Request"}, "id": None})
            
            method = request.get("method")
            params = request.get("params", {})
            req_id = request.get("id")
            
            # Security Envelope Introspection
            meta = request.get("_meta", {})
            authority_sig = meta.get("authority_signature", "")
            payload_string = json.dumps(params, sort_keys=True)
            
            if not self.verify_authority_signature(payload_string, authority_sig):
                return json.dumps({"jsonrpc": "2.0", "error": {"code": -32001, "message": "CRITICAL: Unauthorized Execution Signature for user a.keddeh"}, "id": req_id})

            if method == "tools/call":
                tool_name = params.get("name")
                arguments = params.get("arguments", {})
                
                if tool_name == "mutate_vfs_state":
                    vfs_path = arguments.get("target_vfs_path")
                    digest = arguments.get("payload_digest_sha256")
                    
                    self.vfs_state_registry[vfs_path] = digest
                    logging.info(f"VFS Mutation registered successfully. Path: {vfs_path}, Digest: {digest}")
                    
                    return json.dumps({
                        "jsonrpc": "2.0",
                        "result": {
                            "status": "COMMITTED",
                            "vfs_path": vfs_path,
                            "verified_digest": digest,
                            "authority_lock": "a.keddeh"
                        },
                        "id": req_id
                    })
                
                return json.dumps({"jsonrpc": "2.0", "error": {"code": -32601, "message": "Method not found"}, "id": req_id})
                
            return json.dumps({"jsonrpc": "2.0", "error": {"code": -32601, "message": "Method not found"}, "id": req_id})

        except Exception as e:
            return json.dumps({"jsonrpc": "2.0", "error": {"code": -32603, "message": f"Internal error: {str(e)}"}, "id": None})

    async def client_lifecycle(self, reader: asyncio.StreamReader, writer: asyncio.StreamWriter):
        logging.info("New secure MCP ingress stream initialized.")
        while True:
            data = await reader.readline()
            if not data:
                break
            raw_request = data.decode('utf-8').strip()
            if not raw_request:
                continue
            response = await self.handle_rpc_call(raw_request)
            writer.write((response + "\n").encode('utf-8'))
            await writer.drain()
        writer.close()
        await writer.wait_closed()

    async def start(self):
        server = await asyncio.start_server(self.client_lifecycle, self.host, self.port)
        logging.info(f"Keddeh MCP Core fully deployed on {self.host}:{self.port}")
        async with server:
            await server.serve_forever()
